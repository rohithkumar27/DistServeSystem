# STEP 2 — P1: Arrival rate sweep (canonical DistServe figure)
# --interference-factor 1.3 models the ~30% decode slowdown that
# a colocated GPU suffers when prefill and decode share it.
# Without it, colocated is artificially fast and disagg can't win.
# Runtime: ~25 min
# ------------------------------------------------------------

  --batch-sizes 4,8 \
  --num-requests 32 \
  --arrival-rates 0.0,0.5,1.0,2.0,3.0,4.0 \
  --output-low 8 --output-high 64 \
  --interference-factor 1.3 \
  --out-dir results/report/arrival_sweep
